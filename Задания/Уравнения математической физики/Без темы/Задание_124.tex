\begin{enumerate}[label=\arabic*)]
    \item Re \(\sum_{1}^{\infty } \left(r^ne^{in\varphi }\right)\)
    \item Re \(\frac{re^{in\varphi }}{1-re^{in\varphi }}\)
    \item Re \(\frac{re^{i\varphi }}{1-re^{i\varphi }}\)
    \item \(\frac{r\cos \varphi -r^2}{1-2r\cos \varphi +r^2}\)
    \item \(\frac{r\sin \varphi }{1-2r\cos \varphi +r^2}\)
\end{enumerate}
