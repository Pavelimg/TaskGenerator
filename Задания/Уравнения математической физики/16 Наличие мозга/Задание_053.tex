\begin{enumerate}[label=\arabic*)]
    \item Значение производной функции U в направлении нормали к границе области на границе области
    \item Значение производной функции U в направлении нормали к границе области в
    \item области \(\Omega\)
    \item Значение функции U на границе области \(\Omega\) размерности n
    \item Производную функции U по границе области
\end{enumerate}
