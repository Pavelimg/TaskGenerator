\begin{enumerate}[label=\arabic*)]
    \item нулю
    \item интегралу от этой функции по сфере поделенному на площадь сферы
    \item значению этой функции в центре сферы
    \item среднему арифметическому значению функции на шаре, границей которого является эта сфера
\end{enumerate}
