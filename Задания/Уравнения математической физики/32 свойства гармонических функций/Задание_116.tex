\begin{enumerate}[label=\arabic*)]
    \item нулю
    \item интегралу от этой функции по сфере поделенному на площадь сферы
    \item значению этой функции в центре шара
    \item среднему арифметическому значению функции на границе шара
    \item интегралу от этой функции по шару, поделенному на объём шара
\end{enumerate}
