При замене переменных \(\xi _i=\xi _i(x_1,x_2,\dots ,x_{n)} , i=1,2,\dots , n\) первые производные преобразуются по формуле

\begin{enumerate}[label=\arabic*)]
    \item \(\frac{\partial U}{\partial x_i}\)=\(\sum_{k=1}^{n} \frac{\partial U}{\partial \xi _k}\frac{\partial \xi _i}{\partial x_k}\)
    \item \(\frac{\partial U}{\partial x_i}\)=\(\sum_{k=1}^{n} \frac{\partial U}{\partial \xi _k}\frac{\partial \xi _k}{\partial x_i}\)
    \item \(\frac{\partial U}{\partial x_i}\)=\(\sum_{k=1}^{n} \frac{\partial U}{\partial \xi _k}\) \(\left(\frac{\partial \xi _k}{\partial x_i}\right)^2\)
\end{enumerate}
