    \item \(4\left(\frac{\partial \omega }{\partial x}\right)^2+4\frac{\partial \omega }{\partial x}\frac{\partial \omega }{\partial y}+\left(\frac{\partial \omega }{\partial y}\right)^2+\frac{\partial \omega }{\partial y}=0\)
    \item \(4\left(\frac{\partial \omega }{\partial x}\right)^2+4\frac{\partial \omega }{\partial x}\frac{\partial \omega }{\partial y}+\left(\frac{\partial \omega }{\partial y}\right)^2=0\)
\end{enumerate}
