\begin{enumerate}[label=\arabic*)]
    \item она непрерывна в квадрате \(0\leq x,\xi \leq l\)
    \item она испытывает разрыв на диагонали квадрата \(x = \xi\)
    \item ее первые производные непрерывны в квадрате \(0 \leq x,\xi \leq l\)
    \item ее первые производные испытывают разрыв на диагонали квадрата \(x = \xi\)
\end{enumerate}
